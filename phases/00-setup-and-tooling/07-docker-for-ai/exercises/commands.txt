build:
docker build -t ai-cpu .

run:

docker run --rm -it \
    -v $(pwd):/workspace \
    -v ~/models:/models \
    ai-cpu python -c "import torch; print(f'PyTorch {torch.__version__}, CUDA: {torch.cuda.is_available()}')"

run jupyter:

docker run --rm -it \
    -v $(pwd):/workspace \
    -v ~/models:/models \
    -p 8888:8888 \
    ai-cpu jupyter notebook --ip=0.0.0.0 --port=8888 --no-browser --allow-root


run flask:
docker run --rm -it -v $(pwd):/workspace -p 5000:5000 ai-dev python app.py
